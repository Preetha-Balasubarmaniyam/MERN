3. **Comments in JavaScript:**
- Write a multi-line comment explaining the purpose of a function you create in the next step.
// Example: Multi-line comment with a function in JavaScript
/*
    This function calculates the area of a rectangle.
    Parameters:
      width  - The width of the rectangle (must be a positive number)
      height - The height of the rectangle (must be a positive number)
    Returns:
      The calculated area as a number.
    Note:
      If invalid inputs are provided, the function returns null.
*/
function calculateRectangleArea(width, height) {
    // Validate inputs
    if (typeof width !== "number" || typeof height !== "number" || width <= 0 || height <= 0) {
        console.error("Invalid input: Width and height must be positive numbers.");
        return null;
    }

    // Calculate and return the area
    return width * height;
}

// Example usage:
console.log("Area:", calculateRectangleArea(5, 10)); // Output: Area: 50
console.log("Area:", calculateRectangleArea(-3, 4)); // Output: Error + null
